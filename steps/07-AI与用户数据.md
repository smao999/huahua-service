```python
class AIService:
    """AI 服务——调用大模型 API（OpenRouter 优先，Gemini 降级）"""
    
    def __init__(self):
        self.openrouter_key = settings.OPENROUTER_API_KEY
        self.gemini_key = settings.GEMINI_API_KEY
        # httpx 客户端，支持自定义超时和连接池
        self.http_client = httpx.AsyncClient(
            timeout=httpx.Timeout(settings.OPENROUTER_READ_TIMEOUT, connect=settings.OPENROUTER_CONNECT_TIMEOUT),
            limits=httpx.Limits(max_keepalive_connections=settings.AI_HTTP_POOL_SIZE),
        )
    
    async def chat_completion(self, messages, model=None):
        """发送对话给 AI，先试 OpenRouter，失败降级 Gemini"""
        try:
            return await self._call_openrouter(messages, model)
        except Exception:
            return await self._call_gemini(messages)
    
    async def _call_openrouter(self, messages, model):
        body = {
            "model": model or settings.OPENROUTER_MODEL,
            "messages": [m.dict() for m in messages],
        }
        resp = await self.http_client.post(
            f"{settings.OPENROUTER_BASE_URL}/chat/completions",
            headers={"Authorization": f"Bearer {settings.OPENROUTER_API_KEY}"},
            json=body,
        )
        return resp.json()["choices"][0]["message"]["content"]
```
</details>


**Python 的 Async HTTP 做了什么**：
1. 构造 ChatCompletion 请求体（model + messages）
2. 发 POST 到 OpenRouter API
3. 解析响应，取第一个 choice 的 content
4. 如果 OpenRouter 失败（网络、超时、限流），自动降级到 Gemini

```go
// internal/service/ai/ai_service.go
package ai

import (
    "bytes"
    "context"
    "encoding/json"
    "fmt"
    "io"
    "net/http"
    "time"
)

// AIService AI 对话服务
// 对应 Python AIService
type AIService struct {
    openRouterKey string
    geminiKey     string
    httpClient    *http.Client
}

func NewAIService(openRouterKey, geminiKey string) *AIService {
    return &AIService{
        httpClient: &http.Client{
            Timeout: 90 * time.Second,  // AI 推理慢，给 90 秒
        },
        openRouterKey: openRouterKey,
        geminiKey:     geminiKey,
    }
}

// Message 对话消息
type Message struct {
    Role    string `json:"role"`    // "user" 或 "assistant"
    Content string `json:"content"`
}

// chatRequest OpenRouter 格式的请求体
type chatRequest struct {
    Model    string    `json:"model"`
    Messages []Message `json:"messages"`
}

// chatResponse OpenRouter 格式的响应体
type chatResponse struct {
    Choices []struct {
        Message Message `json:"message"`
    } `json:"choices"`
}

// Chat 发送对话给 AI
// 对应 Python chat_completion()
// 先试 OpenRouter，失败自动降级 Gemini
func (s *AIService) Chat(ctx context.Context, messages []Message) (string, error) {
    // 1. 先试 OpenRouter
    result, err := s.chatWithOpenRouter(ctx, messages)
    if err == nil {
        return result, nil
    }

    // 2. 失败降级 Gemini
    return s.chatWithGemini(ctx, messages)
}

func (s *AIService) chatWithOpenRouter(ctx context.Context, messages []Message) (string, error) {
    if s.openRouterKey == "" {
        return "", fmt.Errorf("OpenRouter 未配置")
    }

    body := chatRequest{
        Model:    "openai/gpt-4o",  // 默认模型
        Messages: messages,
    }
    jsonBody, _ := json.Marshal(body)

    req, err := http.NewRequestWithContext(ctx, "POST",
        "https://openrouter.ai/api/v1/chat/completions",
        bytes.NewReader(jsonBody))
    if err != nil {
        return "", err
    }

    req.Header.Set("Authorization", "Bearer "+s.openRouterKey)
    req.Header.Set("Content-Type", "application/json")

    resp, err := s.httpClient.Do(req)
    if err != nil {
        return "", fmt.Errorf("OpenRouter 请求失败: %w", err)
    }
    defer resp.Body.Close()  // ！必须关，否则连接泄漏

    if resp.StatusCode != http.StatusOK {
        return "", fmt.Errorf("OpenRouter 返回 %d", resp.StatusCode)
    }

    respBody, _ := io.ReadAll(resp.Body)
    var chatResp chatResponse
    if err := json.Unmarshal(respBody, &chatResp); err != nil {
        return "", err
    }

    if len(chatResp.Choices) == 0 {
        return "", fmt.Errorf("AI 未返回结果")
    }

    return chatResp.Choices[0].Message.Content, nil
}

func (s *AIService) chatWithGemini(ctx context.Context, messages []Message) (string, error) {
    // Gemini API 格式略不同，但模式完全一样：
    // 构造请求 → POST → 解析响应
    if s.geminiKey == "" {
        return "", fmt.Errorf("Gemini 未配置")
    }
    return "", fmt.Errorf("Gemini 待实现")
}
```

```python
@router.post("/sync/upload")
async def sync_upload(
    data: SyncData, 
    current_user: User = Depends(get_current_user),
    db: Session = Depends(get_db)
):
    """上传用户基金持仓数据"""
    # 1. 查或创建 user_data 记录
    user_data = db.query(UserData).filter(UserData.user_id == current_user.id).first()
    if not user_data:
        user_data = UserData(user_id=current_user.id)
        db.add(user_data)
    
    # 2. 更新数据
    user_data.data_json = data.json_data
    user_data.data_etag = hashlib.sha256(data.json_data.encode()).hexdigest()[:16]
    user_data.updated_at = datetime.now(timezone.utc).isoformat()
    db.commit()
    
    return {"status": "ok", "etag": user_data.data_etag}

@router.get("/sync/download")
async def sync_download(
    current_user: User = Depends(get_current_user),
    db: Session = Depends(get_db)
):
    """下载用户基金持仓数据"""
    user_data = db.query(UserData).filter(UserData.user_id == current_user.id).first()
    if not user_data:
        return {"data": "{}", "etag": "", "updated_at": ""}
    return {
        "data": user_data.data_json,
        "etag": user_data.data_etag,
        "updated_at": user_data.updated_at,
    }
```
