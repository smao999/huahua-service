## 14.3 基础设施缺失

### 14.3.1 errors/ 包——统一业务错误

Python 的做法是 `raise HTTPException(400, "...")`，Go 需要定义业务错误码，便于前端统一处理。

```go
// internal/errors/errors.go
package errors

import "net/http"

// Error 业务错误——携带 HTTP 状态码和错误码
type Error struct {
    Code       int    `json:"code"`        // 业务码：10000=通用, 10001=参数错误...
    HTTPStatus int    `json:"-"`            // HTTP 状态码（不序列化到 JSON）
    Message    string `json:"detail"`       // 错误消息
}

func (e *Error) Error() string { return e.Message }

// 预定义错误
var (
    ErrUnauthorized      = &Error{Code: 10001, HTTPStatus: http.StatusUnauthorized, Message: "未登录"}
    ErrPermissionDenied  = &Error{Code: 10002, HTTPStatus: http.StatusForbidden, Message: "权限不足"}
    ErrNotFound          = &Error{Code: 10003, HTTPStatus: http.StatusNotFound, Message: "资源不存在"}
    ErrRateLimited       = &Error{Code: 10004, HTTPStatus: http.StatusTooManyRequests, Message: "请求过于频繁"}
    ErrInternal          = &Error{Code: 10500, HTTPStatus: http.StatusInternalServerError, Message: "服务器内部错误"}
)

// NewBadRequest 创建 400 错误
func NewBadRequest(msg string) *Error {
    return &Error{Code: 10400, HTTPStatus: http.StatusBadRequest, Message: msg}
}

// New 自定义业务错误
func New(code int, httpStatus int, msg string) *Error {
    return &Error{Code: code, HTTPStatus: httpStatus, Message: msg}
}
```

然后在全局 Recovery 中间件里统一捕获 `*errors.Error`：

```go
// 在 app.go 的 Recovery 中：
r.Use(func(c *gin.Context) {
    c.Next()
    if len(c.Errors) > 0 {
        for _, e := range c.Errors {
            if appErr, ok := e.Err.(*errors.Error); ok {
                c.AbortWithStatusJSON(appErr.HTTPStatus, appErr)
                return
            }
        }
    }
})
```

### 14.3.2 response/ 包——统一 API 响应格式

Python 直接 `return {"data": ...}`，Go 项目中统一响应格式有两大好处：
1. 前端知道每个响应都有 `code` 和 `message` 字段
2. handler 代码更简洁

```go
// pkg/response/response.go
package response

import (
    "net/http"

    "github.com/gin-gonic/gin"
)

// Response 统一响应结构
type Response struct {
    Code    int         `json:"code"`    // 0=成功，非0=错误码
    Message string      `json:"message"` // 成功="ok"，失败=错误描述
    Data    interface{} `json:"data,omitempty"` // 数据（成功时）
}

// Pagination 分页
type Pagination struct {
    Total int64 `json:"total"`     // 总数
    Page  int   `json:"page"`      // 当前页
    Limit int   `json:"limit"`     // 每页大小
}

// PaginatedData 分页数据
type PaginatedData struct {
    List       interface{} `json:"list"`
    Pagination Pagination  `json:"pagination"`
}

// Success 成功响应
func Success(c *gin.Context, data interface{}) {
    c.JSON(http.StatusOK, Response{Code: 0, Message: "ok", Data: data})
}

// Error 错误响应
func Error(c *gin.Context, httpStatus int, code int, msg string) {
    c.JSON(httpStatus, Response{Code: code, Message: msg})
}

// Paginate 分页响应
func Paginate(c *gin.Context, list interface{}, total int64, page, limit int) {
    c.JSON(http.StatusOK, Response{
        Code:    0,
        Message: "ok",
        Data: PaginatedData{
            List: list,
            Pagination: Pagination{
                Total: total,
                Page:  page,
                Limit: limit,
            },
        },
    })
}
```

**handler 使用示例**：
```go
// 之前：c.JSON(http.StatusOK, gin.H{"data": results, "truncated": false})
// 之后：
response.Success(c, gin.H{"data": results, "truncated": false, "limit": 50})

// 错误时：
response.Error(c, 400, 10400, "参数错误")
```

```go
// internal/middleware/ratelimit.go
package middleware

import (
    "net/http"
    "sync"
    "time"

    "huahua-service/internal/db"
    "huahua-service/internal/pkg/ratelimit"

    "github.com/gin-gonic/gin"
)

// RateLimit 全局 API 限流中间件
// 对应 Python RedisRateLimitMiddleware
// calls: 窗口内允许请求数, period: 窗口大小(秒)
func RateLimit(calls, period int) gin.HandlerFunc {
    // Redis 限流器
    redisLimiter := ratelimit.NewRedisLimiter(db.Redis, calls, period)
    // 进程内降级限流器
    localLimiter := ratelimit.NewLocalLimiter(calls, period)

    return func(c *gin.Context) {
        // 只对 /api/ 路径限流，跳过 OPTIONS
        path := c.Request.URL.Path
        if !startsWith(path, "/api/") || c.Request.Method == "OPTIONS" {
            c.Next()
            return
        }

        key := c.ClientIP()

        // 先试 Redis
        allowed, err := redisLimiter.Allow(c.Request.Context(), key)
        if err == nil && !allowed {
            c.AbortWithStatusJSON(http.StatusTooManyRequests, gin.H{"detail": "请求过于频繁"})
            return
        }

        // Redis 不可用时降级到进程内
        if err != nil {
            if !localLimiter.Allow(key) {
                c.AbortWithStatusJSON(http.StatusTooManyRequests, gin.H{"detail": "请求过于频繁"})
                return
            }
        }

        c.Next()
    }
}

// startsWith 检查字符串前缀
func startsWith(s, prefix string) bool {
    if len(s) < len(prefix) {
        return false
    }
    return s[:len(prefix)] == prefix
}
```

### 14.4.5 BlogStats 中间件（对应 Python）

```go
// internal/middleware/blogstats.go
package middleware

import (
    "crypto/sha256"
    "fmt"
    "time"

    "huahua-service/internal/db"

    "github.com/gin-gonic/gin"
)

// BlogStats 博客访问统计中间件
// 对应 Python BlogVisitStatsMiddleware
// 记录每日访问量、唯一访客数、登录用户数
func BlogStats() gin.HandlerFunc {
    return func(c *gin.Context) {
        c.Next() // 先执行 handler，再统计

        if db.Redis == nil {
            return
        }
        if !shouldTrack(c) {
            return
        }

        ctx := c.Request.Context()
        today := time.Now().Format("2006-01-02")

        // 访问量 +1
        db.Redis.Incr(ctx, fmt.Sprintf("blog_stats:%s:visit_count", today))
        db.Redis.Expire(ctx, fmt.Sprintf("blog_stats:%s:visit_count", today), 72*time.Hour)

        // 唯一访客（SHA256(IP + UA)）
        visitorRaw := c.ClientIP() + "|" + c.GetHeader("User-Agent")
        visitorHash := fmt.Sprintf("%x", sha256.Sum256([]byte(visitorRaw)))
        db.Redis.SAdd(ctx, fmt.Sprintf("blog_stats:%s:visitor_users", today), visitorHash)
        db.Redis.Expire(ctx, fmt.Sprintf("blog_stats:%s:visitor_users", today), 72*time.Hour)
    }
}

// shouldTrack 判断是否需要记录统计：只记录 GET 的 /api/ 路径
func shouldTrack(c *gin.Context) bool {
    path := c.Request.URL.Path
    if c.Request.Method != "GET" || !startsWith(path, "/api/") {
        return false
    }
    // 跳过 blog-stats 自身
    if path == "/api/public/blog-stats" {
        return false
    }
    // 跳过热路径（减少 Redis 写入压力）
    hotPaths := []string{"/api/estimate/batch", "/api/history/", "/api/fund/today-timeline/"}
    for _, hp := range hotPaths {
        if startsWith(path, hp) {
            return false
        }
    }
    return true
}
```

### 14.5.1 pkg/ratelimit/——滑动窗口限流器

```go
// pkg/ratelimit/redis_limiter.go
package ratelimit

import (
    "context"
    "fmt"
    "time"

    "github.com/redis/go-redis/v9"
)

// RedisLimiter Redis 滑动窗口限流器
// 算法：ZREMRANGEBYSCORE 删除过期 → ZCARD 计数 → ZADD + EXPIRE
// 对应 Python security.py 的 check_rate_limit()
type RedisLimiter struct {
    client *redis.Client
    calls  int           // 窗口内允许次数
    period time.Duration // 窗口大小
}

func NewRedisLimiter(client *redis.Client, calls int, period int) *RedisLimiter {
    return &RedisLimiter{
        client: client,
        calls:  calls,
        period: time.Duration(period) * time.Second,
    }
}

// Allow 检查是否允许请求。allowed=false 表示超限
func (l *RedisLimiter) Allow(ctx context.Context, key string) (bool, error) {
    now := time.Now()
    windowStart := now.Add(-l.period)

    pipe := l.client.Pipeline()
    scoreKey := fmt.Sprintf("rl:%s", key)

    // 删除窗口外的条目
    pipe.ZRemRangeByScore(ctx, scoreKey, "0", fmt.Sprintf("%d", windowStart.Unix()))
    // 统计窗口内条数
    countCmd := pipe.ZCard(ctx, scoreKey)
    // 添加当前请求
    pipe.ZAdd(ctx, scoreKey, redis.Z{
        Score:  float64(now.Unix()),
        Member: fmt.Sprintf("%d:%d", now.UnixNano(), time.Now().UnixNano()),
    })
    // 设置过期（窗口 + 1 秒）
    pipe.Expire(ctx, scoreKey, l.period+time.Second)

    _, err := pipe.Exec(ctx)
    if err != nil {
        return false, err
    }

    return countCmd.Val() < int64(l.calls), nil
}
```

```go
// pkg/ratelimit/local_limiter.go
package ratelimit

import (
    "sync"
    "time"
)

// LocalLimiter 进程内滑动窗口限流器（Redis 不可用时降级）
// 对应 Python security.py 的 _local_rate_check()
type LocalLimiter struct {
    mu     sync.Mutex
    windows map[string][]time.Time
    calls  int
    period time.Duration
}

func NewLocalLimiter(calls, period int) *LocalLimiter {
    return &LocalLimiter{
        windows: make(map[string][]time.Time),
        calls:   calls,
        period:  time.Duration(period) * time.Second,
    }
}

// Allow 检查是否允许。goroutine-safe
func (l *LocalLimiter) Allow(key string) bool {
    l.mu.Lock()
    defer l.mu.Unlock()

    now := time.Now()
    cutoff := now.Add(-l.period)

    // 获取或创建窗口
    window, exists := l.windows[key]
    if !exists {
        window = make([]time.Time, 0, l.calls)
    }

    // 删除过期条目
    start := 0
    for i, t := range window {
        if t.After(cutoff) {
            start = i
            break
        }
        start = i + 1
    }
    window = window[start:]

    if len(window) >= l.calls {
        l.windows[key] = window
        return false
    }

    window = append(window, now)
    l.windows[key] = window
    return true
}
```

// pkg/leader/leader.go
package leader

import (
    "context"
    "time"

    "github.com/redis/go-redis/v9"
)

// Lock Redis 分布式锁
// 用于后台任务——多副本部署时只有一个实例执行
type Lock struct {
    client *redis.Client
    key    string
    owner  string
    ttl    time.Duration
}

func NewLock(client *redis.Client, key, owner string, ttl time.Duration) *Lock {
    return &Lock{
        client: client,
        key:    key,
        owner:  owner,
        ttl:    ttl,
    }
}

// Acquire 尝试获取锁。成功返回 true
// 对应 Python cache.acquire_lock(key, ttl=180, owner=leader_id)
func (l *Lock) Acquire(ctx context.Context) (bool, error) {
    return l.client.SetNX(ctx, l.key, l.owner, l.ttl).Result()
}

// Renew 续约锁（心跳）
// 对应 Python cache.acquire_lock（续约用 SET owner = leader_id, KEEPTTL）
func (l *Lock) Renew(ctx context.Context) (bool, error) {
    // 用 Lua 脚本：只有 owner 匹配时才续约
    script := `
    if redis.call("GET", KEYS[1]) == ARGV[1] then
        return redis.call("EXPIRE", KEYS[1], ARGV[2])
    end
    return 0
    `
    result, err := l.client.Eval(ctx, script, []string{l.key}, l.owner, int(l.ttl.Seconds())).Result()
    if err != nil {
        return false, err
    }
    return result.(int64) == 1, nil
}

// Release 释放锁
func (l *Lock) Release(ctx context.Context) error {
    // Lua 脚本：只有 owner 匹配才释放（防止误删他人锁）
    script := `
    if redis.call("GET", KEYS[1]) == ARGV[1] then
        return redis.call("DEL", KEYS[1])
    end
    return 0
    `
    _, err := l.client.Eval(ctx, script, []string{l.key}, l.owner).Result()
    return err
}

// 优先级：X-Forwarded-For → X-Real-IP → Gin RemoteIP
// 对应 Python security.py 的 get_client_ip()
func GetClientIP(c *gin.Context) string {
    // X-Forwarded-For: client, proxy1, proxy2
    forwarded := c.GetHeader("X-Forwarded-For")
    if forwarded != "" {
        parts := strings.Split(forwarded, ",")
        ip := strings.TrimSpace(parts[0])
        if net.ParseIP(ip) != nil {
            return ip
        }
    }

    realIP := c.GetHeader("X-Real-IP")
    if realIP != "" {
        if net.ParseIP(realIP) != nil {
            return realIP
        }
    }

    return c.ClientIP()
}

// Client HTTP 客户端——封装超时、重试、请求头

type Client struct {
    httpClient *http.Client
    headers    map[string]string
}

// Option 客户端配置选项
type Option func(*Client)

func WithTimeout(timeout time.Duration) Option {
    return func(c *Client) {
        c.httpClient.Timeout = timeout
    }
}

func WithHeader(key, value string) Option {
    return func(c *Client) {
        c.headers[key] = value
    }
}

func NewClient(opts ...Option) *Client {
    c := &Client{
        httpClient: &http.Client{
            Timeout: 15 * time.Second,
        },
        headers: make(map[string]string),
    }
    for _, opt := range opts {
        opt(c)
    }
    return c
}

// Get GET 请求
func (c *Client) Get(url string) ([]byte, error) {
    req, err := http.NewRequest("GET", url, nil)
    if err != nil {
        return nil, fmt.Errorf("创建请求失败: %w", err)
    }

    for k, v := range c.headers {
        req.Header.Set(k, v)
    }

    resp, err := c.httpClient.Do(req)
    if err != nil {
        return nil, fmt.Errorf("请求失败: %w", err)
    }
    defer resp.Body.Close()

    if resp.StatusCode != http.StatusOK {
        return nil, fmt.Errorf("HTTP %d", resp.StatusCode)
    }

    body, err := io.ReadAll(resp.Body)
    if err != nil {
        return nil, fmt.Errorf("读取响应失败: %w", err)
    }

    return body, nil
}

// SendVerificationCode 发送验证码邮件
// email: 目标邮箱
// typePrefix: reg/forgot/bind/agent 等
// 对应 Python send_verification_code()
func (s *Service) SendVerificationCode(ctx context.Context, email, typePrefix string) error {
    // 1. 检查发送频率（同一邮箱同一类型 2 分钟冷却）
    rateKey := fmt.Sprintf("mail:rate:%s:%s", typePrefix, email)
    exists, err := s.cache.Get(ctx, rateKey)
    if err == nil && exists != "" {
        return fmt.Errorf("发送太频繁，请稍后再试")
    }

    // 2. 生成 6 位随机验证码
    code, err := generateCode(6)
    if err != nil {
        return err
    }

    // 3. 存到 Redis（10 分钟有效）
    verifyKey := fmt.Sprintf("verify:%s:%s", typePrefix, email)
    if err := s.cache.Set(ctx, verifyKey, code, 10*time.Minute); err != nil {
        return err
    }

    // 4. 设置冷却（2 分钟）
    s.cache.Set(ctx, rateKey, "1", 2*time.Minute)

    // 5. 发邮件（SMTP）
    m := gomail.NewMessage()
    m.SetHeader("From", s.cfg.From)
    m.SetHeader("To", email)
    m.SetHeader("Subject", "花花日记 邮箱验证")
    m.SetBody("text/html", fmt.Sprintf(
        `<div><h2>验证码</h2><p>您的验证码是：<strong style="font-size:24px;color:#2563eb;">%s</strong></p>
         <p>该验证码在 10 分钟内有效。</p></div>`, code))

    d := gomail.NewDialer(s.cfg.Host, s.cfg.Port, s.cfg.Username, s.cfg.Password)
    if err := d.DialAndSend(m); err != nil {
        return fmt.Errorf("邮件发送失败: %w", err)
    }

    return nil
}

// VerifyCode 验证码校验
func (s *Service) VerifyCode(ctx context.Context, email, code, typePrefix string) (bool, error) {
    verifyKey := fmt.Sprintf("verify:%s:%s", typePrefix, email)
    stored, err := s.cache.Get(ctx, verifyKey)
    if err != nil || stored == "" {
        return false, nil
    }
    if stored != code {
        return false, nil
    }
    // 验证成功，删除 key（防止重复使用）
    s.cache.Del(ctx, verifyKey)
    return true, nil
}

func generateCode(length int) (string, error) {
    code := make([]byte, length)
    for i := range code {
        n, err := rand.Int(rand.Reader, big.NewInt(10))
        if err != nil {
            return "", err
        }
        code[i] = byte('0' + n.Int64())
    }
    return string(code), nil
}
