**每个配置项的作用总结分类表**：

| 分类 | 配置项 | 作用 |
|---|---|---|
| 安全 | `SECRET_KEY` | JWT 签名密钥，泄漏 = 任何人都能伪造登录 |
| 安全 | `ADMIN_UID` | 管理员 UID，设置后只有这个 UID 的用户能成为 admin |
| 数据库 | `DATABASE_URL` | 告诉程序"连接到哪个 PostgreSQL" |
| 数据库 | `DB_POOL_SIZE` | 连接池大小——太大浪费内存，太小不够用 |
| Redis | `REDIS_HOST/PORT/PASSWORD/DB` | Redis 连接信息 |
| 限流 | `GLOBAL_RATE_LIMIT_CALLS` | 每个 IP 每分钟最多请求次数，防爬虫 |
| AI | `OPENROUTER_API_KEY` | AI 服务的 API 密钥 |
| 文件 | `DATA_DIR` | 上传头像、截图的存储目录 |
| 邮件 | `MAIL_SERVER/USERNAME/PASSWORD` | 发验证码邮件的 SMTP 配置 |

### Go 实现：internal/config/config.go

Go 没有 pydantic-settings 这种自动读环境变量的库，但我们可以手动实现——反而更清晰。

```go
package config

import (
    "os"
    "strconv"
    "time"
)

// Config 对应 Python 的 Settings 类
// 这是整个 Go 项目唯一的配置结构体
type Config struct {
    // === 应用 ===
    Port     string // 监听端口，默认 "8080"
    Mode     string // 运行模式："debug" 或 "release"
    SecretKey string // JWT 签名密钥
    AdminUID string // 管理员 UID
    
    // === 数据库 ===
    DB     DatabaseConfig
    Redis  RedisConfig
    
    // === 并发 ===
    GlobalRateLimitCalls int // 全局限流：每 IP 每窗口允许请求数
    GlobalRateLimitPeriod int // 限流窗口：秒
    
    // === AI ===
    OpenRouterKey string
    OpenRouterBaseURL string
    GeminiKey string
    
    // === 存储 ===
    DataDir string
    MaxUploadSize int64 // 字节
    
    // === 邮件 ===
    Mail MailConfig
    
    // === 第三方 API ===
    TwelveDataKey string
}

type DatabaseConfig struct {
    DSN          string // PostgreSQL 连接字符串
    MaxOpenConns int    // 最大连接数
    MaxIdleConns int    // 最大空闲连接数
}

type RedisConfig struct {
    Addr     string // host:port
    Password string
    DB       int
}

type MailConfig struct {
    Host     string
    Port     int
    Username string
    Password string
    From     string
}

// Load 从环境变量读取并构造 Config
// 在 main.go 里调用： cfg := config.Load()
func Load() *Config {
    return &Config{
        // 基本
        Port:     getEnv("PORT", "8080"),
        Mode:     getEnv("GIN_MODE", "debug"),
        SecretKey: getEnv("SECRET_KEY", "smartfund_secret_key_change_me"),
        AdminUID: getEnv("ADMIN_UID", ""),
        
        // 数据库
        DB: DatabaseConfig{
            // 注意：Python 的 DATABASE_URL 是 postgresql+psycopg2://...
            // Go 的 GORM 用 postgres://...（去掉 +psycopg2）
            DSN:          getEnv("DATABASE_URL", "postgres://localhost:5432/huahuadaily?sslmode=disable"),
            MaxOpenConns: getEnvInt("DB_POOL_SIZE", 20),
            MaxIdleConns: getEnvInt("DB_MAX_IDLE_CONNS", 10),
        },
        
        // Redis
        Redis: RedisConfig{
            Addr:     getEnv("REDIS_ADDR", "localhost:6379"),
            Password: getEnv("REDIS_PASSWORD", ""),
            DB:       getEnvInt("REDIS_DB", 0),
        },
        
        // 限流
        GlobalRateLimitCalls: getEnvInt("GLOBAL_RATE_LIMIT_CALLS", 1200),
        GlobalRateLimitPeriod: getEnvInt("GLOBAL_RATE_LIMIT_PERIOD", 60),
        
        // AI
        OpenRouterKey:    getEnv("OPENROUTER_API_KEY", ""),
        OpenRouterBaseURL: getEnv("OPENROUTER_BASE_URL", "https://openrouter.ai/api/v1"),
        GeminiKey:        getEnv("GEMINI_API_KEY", ""),
        
        // 文件
        DataDir:       getEnv("DATA_DIR", "./data"),
        MaxUploadSize: int64(getEnvInt("MAX_UPLOAD_SIZE", 5*1024*1024)),
        
        // 邮件
        Mail: MailConfig{
            Host:     getEnv("MAIL_SERVER", "smtp.qq.com"),
            Port:     getEnvInt("MAIL_PORT", 465),
            Username: getEnv("MAIL_USERNAME", ""),
            Password: getEnv("MAIL_PASSWORD", ""),
            From:     getEnv("MAIL_FROM", ""),
        },
        
        TwelveDataKey: getEnv("TWELVE_DATA_API_KEY", ""),
    }
}

// getEnv 读取环境变量，不存在用默认值
func getEnv(key, defaultVal string) string {
    if v := os.Getenv(key); v != "" {
        return v
    }
    return defaultVal
}

// getEnvInt 读取整数环境变量
func getEnvInt(key string, defaultVal int) int {
    if v := os.Getenv(key); v != "" {
        if n, err := strconv.Atoi(v); err == nil {
            return n
        }
    }
    return defaultVal
}

// 注意：
// 有些配置 Python 放在 Settings 里但 Go 暂时不需要，
// 比如 TRUSTED_PROXY_COUNT、CORS_ORIGINS 后面在中间件里处理
// AKSHARE_*_CONCURRENCY 等并发控制在 service 层处理
```

## 1.2 database.py → Go db

### Python 源码：database.py

```python
# database.py — SQLAlchemy 数据库连接
from sqlalchemy import create_engine
from sqlalchemy.ext.declarative import declarative_base
from sqlalchemy.orm import sessionmaker
from config import settings

# 创建数据库引擎
engine = create_engine(
    settings.DATABASE_URL,
    # 连接池配置
    pool_size=settings.DB_POOL_SIZE,     # 池子里保持 20 个连接
    max_overflow=settings.DB_MAX_OVERFLOW, # 高峰期最多额外创建 40 个
    pool_timeout=30,     # 等池子空闲连接的最大秒数
    pool_recycle=1800,   # 连接超过 30 分钟自动重建（防连接断开）
    pool_pre_ping=True,  # 取连接前先 PING 一下，断了的自动重建
)

# SessionLocal 是"会话工厂"，每次请求用它创建一个数据库会话
SessionLocal = sessionmaker(autocommit=False, autoflush=False, bind=engine)

# Base 是所有模型类的基类
# SQLAlchemy 通过它知道表结构定义
Base = declarative_base()


def get_db():
    """FastAPI 的依赖注入函数。
    每次请求时调用，请求结束自动关闭连接。
    用法：def handler(db: Session = Depends(get_db)):
    """
    db = SessionLocal()
    try:
        yield db   # 把 db 传给路由函数
    finally:
        db.close()  # 保证连接一定会归还到池子
```

**database.py 做了什么？**
1. 创建 PostgreSQL 连接池（连接不真"断开"，而是重用）
2. 定义了 `get_db()` 依赖——每次请求创建一个会话，请求结束自动关闭
3. `Base` 是所有 ORM 模型的父类

### Go 实现：internal/db/db.go

```go
package db

import (
    "context"
    "fmt"
    "log/slog"
    "time"

    "huahua-service/internal/config"

    "github.com/redis/go-redis/v9"
    "gorm.io/driver/postgres"
    "gorm.io/gorm"
    "gorm.io/gorm/logger"
)

// 全局变量——整个 Go 程序都用这两个对象操作数据库
var (
    GORM  *gorm.DB
    Redis *redis.Client
)

// InitGORM 初始化 GORM 数据库连接
// 对应 Python create_engine() + SessionLocal()
func InitGORM(cfg config.DatabaseConfig) {
    // GORM 的 Open 等价于 SQLAlchemy 的 create_engine
    // postgres.Open 接收 DSN
    db, err := gorm.Open(postgres.Open(cfg.DSN), &gorm.Config{
        // GORM 日志级别
        Logger: logger.Default.LogMode(logger.Info),
    })
    if err != nil {
        // 启动阶段连不上数据库就直接 panic——早死早超生
        panic(fmt.Sprintf("数据库连接失败: %v", err))
    }

    // 从 GORM 获取底层连接池（*sql.DB）
    // 然后配置连接池参数
    sqlDB, err := db.DB()
    if err != nil {
        panic(fmt.Sprintf("获取底层 DB 失败: %v", err))
    }

    // 对应 Python 的 pool_size + max_overflow
    sqlDB.SetMaxOpenConns(cfg.MaxOpenConns)     // 最大打开连接数
    sqlDB.SetMaxIdleConns(cfg.MaxIdleConns)     // 最大空闲连接数
    sqlDB.SetConnMaxLifetime(30 * time.Minute)  // 对应 pool_recycle=1800

    GORM = db
    slog.Info("数据库已连接", "max_open", cfg.MaxOpenConns)
}

// InitRedis 初始化 Redis 连接
// 对应 Python cache.py 的 Cache.__init__
func InitRedis(cfg config.RedisConfig) {
    client := redis.NewClient(&redis.Options{
        Addr:         cfg.Addr,
        Password:     cfg.Password,
        DB:           cfg.DB,
        DialTimeout:  5 * time.Second,  // 连接超时
        ReadTimeout:  3 * time.Second,  // 读超时
        WriteTimeout: 3 * time.Second,  // 写超时
    })

    // PING 测试连接——就像 Python cache.py 里用 redis.ping()
    ctx, cancel := context.WithTimeout(context.Background(), 5*time.Second)
    defer cancel()

    if err := client.Ping(ctx).Err(); err != nil {
        panic(fmt.Sprintf("Redis 连接失败: %v", err))
    }

    Redis = client
    slog.Info("Redis 已连接", "addr", cfg.Addr)
}

// Close 关闭所有连接——程序退出时调用
func Close() {
    if GORM != nil {
        if sqlDB, err := GORM.DB(); err == nil {
            sqlDB.Close()
        }
    }
    if Redis != nil {
        Redis.Close()
    }
}
```

```python
# models.py — 所有数据库表的 ORM 定义

from sqlalchemy import Column, Integer, String, Text, Boolean, DateTime, Float, Index, Numeric
from database import Base
from datetime import datetime, timezone
import secrets, string


def _utcnow() -> datetime:
    """返回当前 UTC 时间（没有时区信息）"""
    return datetime.now(timezone.utc).replace(tzinfo=None)


def generate_uid():
    """生成 8 位数字 UID"""
    return ''.join(secrets.choice(string.digits) for _ in range(8))


# ──────── 第 1 张表：users ────────
class User(Base):
    __tablename__ = "users"
    id            = Column(Integer, primary_key=True, index=True)
    username      = Column(String, unique=True, index=True)
    email         = Column(String, unique=True, index=True, nullable=True)
    hashed_password = Column(String)                    # bcrypt 哈希
    avatar        = Column(String, default="")
    nickname      = Column(String, default="")
    uid           = Column(String, unique=True, index=True, default=generate_uid)
    invited_by    = Column(Integer, nullable=True)       # 邀请人 user.id（可 null）
    vip_expires_at = Column(DateTime, default=lambda: datetime(2000, 1, 1))
    pro_expires_at = Column(DateTime, nullable=True)     # 高级会员到期
    is_admin      = Column(Boolean, default=False)
    created_at    = Column(DateTime, default=_utcnow)

# ──────── 第 2 张表：user_data（用户基金数据备份）───────
class UserData(Base):
    __tablename__ = "user_data"
    id        = Column(Integer, primary_key=True)
    user_id   = Column(Integer, unique=True, index=True)
    data_json = Column(Text)          # 基金持仓 JSON
    data_etag = Column(String(32))    # 数据变更校验码
    updated_at = Column(String)

# ──────── 第 3 张表：fund_navs（基金历史净值）───────
class FundNav(Base):
    __tablename__ = "fund_navs"
    id        = Column(Integer, primary_key=True)
    fund_code = Column(String, index=True)
    date      = Column(String, index=True)      # "2024-01-15"
    nav       = Column(Numeric(10, 4))           # 单位净值
    change    = Column(Numeric(7, 4))            # 涨跌幅

# ──────── 第 4 张表：fund_sectors（基金行业分类）───────
class FundSector(Base):
    __tablename__ = "fund_sectors"
    id         = Column(Integer, primary_key=True)
    fund_code  = Column(String, unique=True)
    sector     = Column(String)
    updated_at = Column(DateTime)

# ──────── 第 5 张表：app_versions（应用版本管理）───────
class AppVersion(Base):
    __tablename__ = "app_versions"
    id           = Column(Integer, primary_key=True)

    created_at   = Column(DateTime)

# ──────── 第 6 张表：fund_basic_info（基金基本信息）───────
class FundBasicInfo(Base):
    __tablename__ = "fund_basic_info"
    code          = Column(String, primary_key=True)
    name          = Column(String)
    type          = Column(String)          # "股票型" "混合型" ...
```

```python
# cache.py — Redis 缓存 + 进程内 LRU 降级

import redis.asyncio as redis
import json
from collections import OrderedDict
from config import settings

class DateTimeEncoder(json.JSONEncoder):
    """JSON 编码器：处理 datetime 和 Decimal 类型"""
    def default(self, obj):
        if isinstance(obj, (datetime.datetime, datetime.date)):
            return obj.isoformat()
        if isinstance(obj, Decimal):
            return float(obj)
        return super().default(obj)


# ─── 进程内内存缓存（Redis 不可用时的降级方案）───
_MEM_MAX_KEYS = 2000
_mem: OrderedDict = OrderedDict()

def _mem_get(key):
    entry = _mem.get(key)
    if entry is None:
        return None
    if entry["exp"] != -1 and entry["exp"] < time.monotonic():
        _mem.pop(key, None)  # 惰性过期：读取时发现超时就删除
        return None
    _mem.move_to_end(key)    # LRU 策略：把命中的 key 移到末尾
    return entry["val"]

def _mem_set(key, value, ttl=300):
    exp = -1 if ttl == -1 else time.monotonic() + ttl
    while len(_mem) >= _MEM_MAX_KEYS:
        _mem.popitem(last=False)   # 淘汰最久的 key
    _mem[key] = {"val": value, "exp": exp}


class Cache:
    """统一缓存接口：优先 Redis，不可用时自动降级到进程内内存"""
    
    def __init__(self):
        self.redis = redis.Redis(
            host=settings.REDIS_HOST,
            port=settings.REDIS_PORT,
            password=settings.REDIS_PASSWORD,
            db=settings.REDIS_DB,
            socket_timeout=settings.REDIS_SOCKET_TIMEOUT,
            max_connections=settings.REDIS_MAX_CONNECTIONS,
        )
        self._redis_available = True
    
    async def get(self, key):
        try:
            return await self.redis.get(key)
        except Exception:
            return _mem_get(key)
    
    async def set(self, key, value, ttl=300):
        try:
            await self.redis.set(key, value, ex=ttl)
        except Exception:
            _mem_set(key, value, ttl)
    
    async def delete(self, key):
        try:
            await self.redis.delete(key)
        except Exception:
            _mem_delete(key)

cache = Cache()  # 全局单例
```

**cache.py 的作用**：
- 封装 Redis 操作（get/set/delete）
- 当 Redis 宕机时自动降级到进程内 LRU 缓存（最大 2000 个 key）
- 惰性过期 + LRU 淘汰策略防止 OOM

### Go 实现：pkg/cache/cache.go

```go
package cache

import (
    "context"
    "encoding/json"
    "sync"
    "time"

    "container/list"

    "github.com/redis/go-redis/v9"
)

// Cache 缓存层——优先 Redis，不可用时进程内 LRU
type Cache struct {
    client     *redis.Client
    local      *localCache    // 内存降级缓存
    redisAvail bool
    mu         sync.RWMutex
}

// localEntry 内存缓存条目
type localEntry struct {
    value  string
    expiry int64 // unix nano，-1 表示永不过期
}

// localCache 简单的 LRU 缓存
type localCache struct {
    maxKeys int
    ll      *list.List
    items   map[string]*list.Element
    mu      sync.Mutex
}

type cacheItem struct {
    key   string
    entry localEntry
}

func NewLocalCache(maxKeys int) *localCache {
    return &localCache{
        maxKeys: maxKeys,
        ll:      list.New(),
        items:   make(map[string]*list.Element),
    }
}

func (c *localCache) get(key string) (string, bool) {
    c.mu.Lock()
    defer c.mu.Unlock()

    el, ok := c.items[key]
    if !ok {
        return "", false
    }

    entry := el.Value.(*cacheItem).entry
    // 检查过期
    if entry.expiry != -1 && time.Now().UnixNano() > entry.expiry {
        c.ll.Remove(el)
        delete(c.items, key)
        return "", false
    }

    // LRU：移动到前面
    c.ll.MoveToFront(el)
    return entry.value, true
}

func (c *localCache) set(key, value string, ttl time.Duration) {
    c.mu.Lock()
    defer c.mu.Unlock()

    var expiry int64 = -1
    if ttl > 0 {
        expiry = time.Now().Add(ttl).UnixNano()
    }

    // 如果 key 已存在，更新并移到前面
    if el, ok := c.items[key]; ok {
        el.Value.(*cacheItem).entry = localEntry{value: value, expiry: expiry}
        c.ll.MoveToFront(el)
        return
    }

    // 淘汰：超过最大 key 数时删除最旧的
    for c.ll.Len() >= c.maxKeys {
        back := c.ll.Back()
        if back == nil {
            break
        }
        c.ll.Remove(back)
        delete(c.items, back.Value.(*cacheItem).key)
    }

    el := c.ll.PushFront(&cacheItem{key: key, entry: localEntry{value: value, expiry: expiry}})
    c.items[key] = el
}

// NewCache 创建缓存实例
func NewCache(client *redis.Client) *Cache {
    return &Cache{
        client:     client,
        local:      NewLocalCache(2000),
        redisAvail: client != nil,
    }
}

// Get 取值
func (c *Cache) Get(ctx context.Context, key string) (string, error) {
    // 先试 Redis
    if c.redisAvail && c.client != nil {
        val, err := c.client.Get(ctx, key).Result()
        if err == nil {
            return val, nil
        }
        if err != redis.Nil {
            // Redis 出错，标记不可用
            c.mu.Lock()
            c.redisAvail = false
            c.mu.Unlock()
        }
    }

    // 降级到内存缓存
    val, ok := c.local.get(key)
    if !ok {
        return "", redis.Nil
    }
    return val, nil
}

// Set 设值
func (c *Cache) Set(ctx context.Context, key string, value interface{}, ttl time.Duration) error {
    var strVal string
    switch v := value.(type) {
    case string:
        strVal = v
    default:
        b, _ := json.Marshal(v)
        strVal = string(b)
    }

    if c.redisAvail && c.client != nil {
        err := c.client.Set(ctx, key, strVal, ttl).Err()
        if err == nil {
            return nil
        }
        c.mu.Lock()
        c.redisAvail = false
        c.mu.Unlock()
    }

    // 降级
    c.local.set(key, strVal, ttl)
    return nil
}

// Del 删值
func (c *Cache) Del(ctx context.Context, key string) error {
    if c.redisAvail && c.client != nil {
        return c.client.Del(ctx, key).Err()
    }
    return nil
}
```

```go
// internal/security/rate_limit.go
package security

import (
    "context"
    "fmt"
    "time"

    "github.com/redis/go-redis/v9"
)

// RateLimiter 滑动窗口限流器
// 对应 Python check_rate_limit()
type RateLimiter struct {
    redis *redis.Client
}

func NewRateLimiter(redis *redis.Client) *RateLimiter {
    return &RateLimiter{redis: redis}
}

// Check 检查是否超过限流
// keySuffix 如 "ip:192.168.1.1:register"
// limit 允许的最大次数
// period 时间窗口（秒）
// 返回 nil 表示允许请求，error 表示超限
//
// 算法原理（和 Python 一模一样）：
// 1. ZREMRANGEBYSCORE 删除窗口外的条目
// 2. ZCARD 统计窗口内数量
// 3. 超过 limit 返回错误
// 4. ZADD 新增当前请求
// 5. EXPIRE 设置过期时间
func (rl *RateLimiter) Check(ctx context.Context, namespace, keySuffix string, limit, period int) error {
    key := fmt.Sprintf("rl:%s:%s", namespace, keySuffix)
    now := time.Now().Unix()
    windowStart := now - int64(period)

    pipe := rl.redis.Pipeline()
    pipe.ZRemRangeByScore(ctx, key, "0", fmt.Sprintf("%d", windowStart))
    pipe.ZCard(ctx, key)
    pipe.ZAdd(ctx, key, redis.Z{
        Score:  float64(now),
        Member: fmt.Sprintf("%d:%d", now, time.Now().UnixNano()),
    })
    pipe.Expire(ctx, key, time.Duration(period+1)*time.Second)

    cmds, err := pipe.Exec(ctx)
    if err != nil {
        // Redis 挂了——目前简单放行，后续可以加内存降级
        return nil
    }

    count := cmds[1].(*redis.IntCmd).Val()
    if count >= int64(limit) {
        return fmt.Errorf("too many requests")
    }

    return nil
}

// Cooldown 冷却期检查
// key 如 "cooldown:send-code:test@test.com:REGISTER"
// 如果 key 存在，说明还在冷却期内
// 如果 key 不存在，设置 key 并开始计时
func (rl *RateLimiter) Cooldown(ctx context.Context, key string, ttl int) error {
    exists, err := rl.redis.Exists(ctx, key).Result()
    if err != nil {
        return nil // Redis 挂了，放行
    }
    if exists > 0 {
        return fmt.Errorf("cooldown period not expired")
    }
    return rl.redis.Set(ctx, key, "1", time.Duration(ttl)*time.Second).Err()
}
```

```python
# utils.py — 工具函数

from datetime import datetime, timezone
from decimal import Decimal, InvalidOperation, ROUND_HALF_UP


def utcnow() -> datetime:
    """返回当前 UTC 时间（无时区信息），与数据库存储格式一致"""
    return datetime.now(timezone.utc).replace(tzinfo=None)


def round_half_up(value, digits: int = 0):
    """四舍五入（半向上）——和 Python 内置 round() 不同！
    
    Python 内置 round() 是"银行家舍入"（0.5 向偶舍入）
    而金融计算需要用"半向上"（0.005 → 0.01）
    
    例子：
    round_half_up(1.2345, 2) → 1.23
    round_half_up(1.2350, 2) → 1.24
    """
    try:
        dec = Decimal(str(value))
        if not dec.is_finite():
            return 0 if digits == 0 else 0.0
        quant = Decimal("1") if digits == 0 else Decimal("1").scaleb(-digits)
        rounded = dec.quantize(quant, rounding=ROUND_HALF_UP)
    except (InvalidOperation, TypeError, ValueError):
        return 0 if digits == 0 else 0.0
    return int(rounded) if digits == 0 else float(rounded)
```

```go
import "github.com/shopspring/decimal"

// RoundHalfUp 半向上舍入
// 对应 Python round_half_up()
//
// Python 的 round() 是银行家舍入（0.5 舍向偶），
// Go 的 decimal.Round() 默认也是银行家舍入，

//

//   result := util.RoundHalfUp(decimal.NewFromFloat(1.235), 2)

func RoundHalfUp(v decimal.Decimal, digits int32) decimal.Decimal {
    // decimal.New(1, -digits) 创建 10^(-digits)，如 digits=2 则创建 0.01
    return v.Quantize(decimal.New(1, -digits), decimal.RoundHalfUp)
```

```go
    ctx, cancel := context.WithTimeout(context.Background(), 10*time.Second)
    defer cancel()
```

## 补充 1：errors/ 包——统一业务错误（从 Step 14 移入）

Python 的做法是 `raise HTTPException(400, "...")`，Go 里需要定义业务错误码，这样前端可以统一处理。

```go
// internal/errors/errors.go — 新建文件
package errors

import "net/http"

// Error 业务错误——携带 HTTP 状态码和业务码
type Error struct {
    Code       int    `json:"code"`        // 业务码：10000=通用, 10001=参数错误, 10500=服务器错误
    HTTPStatus int    `json:"-"`            // HTTP 状态码（不序列化到 JSON 响应中）
    Message    string `json:"detail"`       // 错误描述
}

func (e *Error) Error() string { return e.Message }

// 预定义业务错误——所有 handler 和 service 共用
var (
    ErrUnauthorized     = &Error{Code: 10001, HTTPStatus: http.StatusUnauthorized, Message: "未登录"}
    ErrPermissionDenied = &Error{Code: 10002, HTTPStatus: http.StatusForbidden, Message: "权限不足"}
    ErrNotFound         = &Error{Code: 10003, HTTPStatus: http.StatusNotFound, Message: "资源不存在"}
    ErrRateLimited      = &Error{Code: 10004, HTTPStatus: http.StatusTooManyRequests, Message: "请求过于频繁"}
    ErrBadRequest       = &Error{Code: 10400, HTTPStatus: http.StatusBadRequest, Message: "参数错误"}
    ErrInternal         = &Error{Code: 10500, HTTPStatus: http.StatusInternalServerError, Message: "服务器内部错误"}
)

// NewBadRequest 创建 400 错误——自定义消息
func NewBadRequest(msg string) *Error {
    return &Error{Code: 10400, HTTPStatus: http.StatusBadRequest, Message: msg}
}

// New 自定义业务错误
func New(code int, httpStatus int, msg string) *Error {
    return &Error{Code: code, HTTPStatus: httpStatus, Message: msg}
}
```

## 补充 4：pkg/ratelimit/——滑动窗口限流器（从 Step 14 移入）

供中间件和 security 层使用的独立限流工具。

```go
// pkg/ratelimit/redis_limiter.go — 新建文件
package ratelimit

import (
    "context"
    "fmt"
    "time"
    "github.com/redis/go-redis/v9"
)

// RedisLimiter Redis 滑动窗口限流器
// 算法：ZREMRANGEBYSCORE 删除过期 → ZCARD 计数 → ZADD + EXPIRE
type RedisLimiter struct {
    client *redis.Client
    calls  int
    period time.Duration
}

func NewRedisLimiter(client *redis.Client, calls int, periodSec int) *RedisLimiter {
    return &RedisLimiter{client: client, calls: calls, period: time.Duration(periodSec) * time.Second}
}

func (l *RedisLimiter) Allow(ctx context.Context, key string) (bool, error) {
    now := time.Now()
    windowStart := now.Add(-l.period)
    scoreKey := fmt.Sprintf("rl:%s", key)
    pipe := l.client.Pipeline()
    pipe.ZRemRangeByScore(ctx, scoreKey, "0", fmt.Sprintf("%d", windowStart.Unix()))
    countCmd := pipe.ZCard(ctx, scoreKey)
    pipe.ZAdd(ctx, scoreKey, redis.Z{Score: float64(now.Unix()), Member: fmt.Sprintf("%d", now.UnixNano())})
    pipe.Expire(ctx, scoreKey, l.period+time.Second)
    _, err := pipe.Exec(ctx)
    if err != nil { return true, err }
    return countCmd.Val() < int64(l.calls), nil
}
```

```go
// pkg/ratelimit/local_limiter.go — 新建文件（Redis 不可用时降级）
package ratelimit

import "sync"
import "time"

type LocalLimiter struct {
    mu      sync.Mutex
    windows map[string][]time.Time
    calls   int
    period  time.Duration
}

func NewLocalLimiter(calls, periodSec int) *LocalLimiter {
    return &LocalLimiter{windows: make(map[string][]time.Time), calls: calls, period: time.Duration(periodSec) * time.Second}
}

func (l *LocalLimiter) Allow(key string) bool {
    l.mu.Lock()
    defer l.mu.Unlock()
    now := time.Now()
    cutoff := now.Add(-l.period)
    window := l.windows[key]
    // 清过期
    var valid []time.Time
    for _, t := range window {
        if t.After(cutoff) { valid = append(valid, t) }
    }
    if len(valid) >= l.calls { l.windows[key] = valid; return false }
    valid = append(valid, now)
    l.windows[key] = valid
    return true
}
```

## 补充 5：pkg/ipx/——IP 提取工具（从 Step 14 移入）

```go
// pkg/ipx/ip.go — 新建文件
package ipx

import (
    "net"
    "strings"
    "github.com/gin-gonic/gin"
)

// GetClientIP 提取客户端真实 IP
// 优先级：X-Forwarded-For → X-Real-IP → Gin RemoteIP
func GetClientIP(c *gin.Context) string {
    forwarded := c.GetHeader("X-Forwarded-For")
    if forwarded != "" {
        parts := strings.Split(forwarded, ",")
        ip := strings.TrimSpace(parts[0])
        if net.ParseIP(ip) != nil { return ip }
    }
    realIP := c.GetHeader("X-Real-IP")
    if realIP != "" {
        if net.ParseIP(realIP) != nil { return realIP }
    }
    return c.ClientIP()
}
```
