# Step 9：后台任务

> **目标**：写一个后台循环，定时刷新基金数据、执行夜估。
> 这个循环和 Gin 服务**同时启动**，在另一个 goroutine 中运行，互不干扰。

---

## 9.1 Python 源码分析

**tasks/background_refresh.py**（~200 行）：

```python
"""统一后台轮询任务（所有时间为北京时间）

任务清单（仅交易日执行）：
① 全量基金列表  — 每天 04:00 刷新
② 板块排行      — 交易日 9:15-15:05 每 10min
③ 基金涨跌排行  — 交易日 20:00-24:00 每 1h
④ 全量净值      — 交易日 17:00-23:50 每 5min
⑨ 今日实时涨跌榜 — 交易日 9:30-15:05 每 5min
⑩ 盘后涨跌榜兜底 — 交易日 17:00-21:00 每 5min
"""

_BASE_INTERVAL = 60  # 心跳间隔 60 秒
_LEADER_LOCK_KEY = "lock:bg_refresh_loop:v1"
_LEADER_LOCK_TTL = 180

async def background_refresh_loop():
    """主循环：每 60 秒检查一次该做什么"""
    while True:
        try:
            now = get_beijing_time()
            
            # 非交易日跳过（节省资源）
            if not is_trading_day(now):
                await asyncio.sleep(300)  # 非交易日 5 分钟检查一次
                continue
            
            # 检查各个时间窗口
            hour, minute = now.hour, now.minute
            
            # 04:00 — 刷新基金列表
            if hour == 4 and minute == 0:
                await refresh_fund_list()
            
            # 9:15-15:05 — 板块排行
            if (hour == 9 and minute >= 15) or (10 <= hour <= 14) or (hour == 15 and minute <= 5):
                if minute % 10 == 0:
                    await refresh_sector_rank()
            
            # 22:00 — 夜估校准
            if hour == 22 and minute == 0:
                await CalibrationService.run_calibration(...)
        
        except Exception as e:
            logger.error(f"后台任务异常: {e}")
        
        await asyncio.sleep(_BASE_INTERVAL)
```

**Python 后台任务的关键设计**：
1. 用 `asyncio.sleep` 做定时——不在 sleep 时占用 CPU
2. 用 Redis 分布式锁确保"只有一个实例在跑"（多副本部署时）
3. 非交易日大幅降低检查频率（5 分钟一次）
4. 每个子任务独立 try/except——一个失败不影响其他

## 9.2 Go 实现

### 9.2.1 后台任务运行器

```go
// internal/task/background.go
package task

import (
    "context"
    "log/slog"
    "time"
)

// Runner 后台任务运行器
// 对应 Python background_refresh_loop()
type Runner struct {
    calibService *CalibrationService  // 需要从外部传入
}

// Config 后台任务配置
type Config struct {
    CheckInterval time.Duration // 心跳间隔（默认 60 秒）
}

func NewRunner(cfg Config) *Runner {
    return &Runner{}
}

// Run 启动后台循环
// 在 goroutine 中调用：go taskRunner.Run(ctx)
// ctx 用来优雅退出——收到取消信号时停止循环
func (r *Runner) Run(ctx context.Context) {
    slog.Info("后台任务启动")
    ticker := time.NewTicker(60 * time.Second)
    defer ticker.Stop()

    for {
        select {
        case <-ticker.C:
            r.tick(ctx)
        case <-ctx.Done():
            slog.Info("后台任务停止")
            return
        }
    }
}

// tick 每 60 秒执行一次的任务检查
func (r *Runner) tick(ctx context.Context) {
    now := time.Now()
    
    // 非交易日跳过（简单判断：周末）
    if now.Weekday() == time.Saturday || now.Weekday() == time.Sunday {
        return
    }

    hour, min := now.Hour(), now.Minute()

    switch {
    case hour == 4 && min == 0:
        // 04:00 — 刷新基金全列表
        slog.Info("刷新基金列表...")
    
    case (hour == 9 && min >= 15) || (hour >= 10 && hour <= 14) || (hour == 15 && min <= 5):
        // 盘中：每 10 分钟刷新板块排行
        if min%10 == 0 {
            slog.Info("刷新板块排行...")
        }
    
    case hour == 22 && min == 0:
        // 22:00 — 夜估校准
        slog.Info("执行夜估校准...")
        if r.calibService != nil {
            date := now.Format("2006-01-02")
            if err := r.calibService.RunCalibration(ctx, date); err != nil {
                slog.Error("夜估失败", "error", err)
            }
        }
    }
}
```

import "huahua-service/internal/task"

func main() {

    db.InitGORM(cfg.DB)
    db.InitRedis(cfg.Redis)
    defer db.Close()

            slog.Error("服务异常", "error", err)
            os.Exit(1)
        }
    }()

    // 优雅退出顺序：先停 HTTP → 再停后台 → 最后关数据库
    <-quit
    slog.Info("正在关闭...")

    bgCancel() // 通知后台任务退出
    db.Close()
}

### 9.2.3 Go 的 select + ticker 模式

```go
// select + ticker 是 Go 实现"定时 + 可取消"循环的标准模式

ticker := time.NewTicker(60 * time.Second)
defer ticker.Stop()

for {
    select {
    case <-ticker.C:
        // 每 60 秒执行一次
        doWork()
    case <-ctx.Done():
        // 收到取消信号，退出循环
        return
    }
}
```

**对比 Python**：
- Python: `while True: await asyncio.sleep(60); doWork()`
- Go: `select { case <-ticker.C: doWork(); case <-ctx.Done(): return }`

Go 的优势：`ctx.Done()` 保证了"收到取消信号时能立即退出"。
Python 的 `asyncio.sleep(60)` 在 sleep 的 60 秒内收到取消信号是**不会立即退出**的——必须等 sleep 结束。

## 补充：Redis 分布式领导锁（从 Step 14 移入）

多副本部署时，后台任务**只能由一个实例执行**，否则会有重复刷新、重复夜估等问题。
Python 用 `cache.acquire_lock(key, ttl=180, owner=leader_id)` 实现。

### Python 源码

```python
# tasks/background_refresh.py
_LEADER_LOCK_KEY = "lock:bg_refresh_loop:v1"
_LEADER_LOCK_TTL = 180
_LEADER_HEARTBEAT_INTERVAL = 60

async def background_refresh_loop():
    leader_id = f"{os.getpid()}:{uuid4().hex}"
    
    while True:
        # 1. 尝试获取锁（SET NX，TTL=180）
        acquired = await cache.acquire_lock(
            _LEADER_LOCK_KEY, ttl=_LEADER_LOCK_TTL, owner=leader_id
        )
        
        if not acquired:
            # 不是 leader，休眠后重试
            await asyncio.sleep(60)
            continue
        
        # 2. 启动心跳 goroutine（每 60 秒续约一次）
        start_heartbeat()
        
        # 3. 执行任务
        while True:
            try:
                # ... 执行所有定时任务 ...
            except Exception:
                pass
            await asyncio.sleep(60)
```

### Go 实现

```go
// pkg/leader/leader.go — 新建文件
package leader

import (
    "context"
    "time"
    "github.com/redis/go-redis/v9"
)

// Lock Redis 分布式锁——用于后台任务领导选举
type Lock struct {
    client *redis.Client
    key    string
    owner  string
    ttl    time.Duration
}

func NewLock(client *redis.Client, key, owner string, ttl time.Duration) *Lock {
    return &Lock{client: client, key: key, owner: owner, ttl: ttl}
}

// Acquire 尝试获取锁（SET NX + EX）
func (l *Lock) Acquire(ctx context.Context) (bool, error) {
    return l.client.SetNX(ctx, l.key, l.owner, l.ttl).Result()
}

// Renew 续约锁（只有 owner 匹配时才续约）
func (l *Lock) Renew(ctx context.Context) (bool, error) {
    script := `if redis.call("GET", KEYS[1]) == ARGV[1] then return redis.call("EXPIRE", KEYS[1], ARGV[2]) else return 0 end`
    result, err := l.client.Eval(ctx, script, []string{l.key}, l.owner, int(l.ttl.Seconds())).Result()
    if err != nil { return false, err }
    return result.(int64) == 1, nil
}

// Release 释放锁（只有 owner 匹配才释放）
func (l *Lock) Release(ctx context.Context) error {
    script := `if redis.call("GET", KEYS[1]) == ARGV[1] then return redis.call("DEL", KEYS[1]) else return 0 end`
    _, err := l.client.Eval(ctx, script, []string{l.key}, l.owner).Result()
    return err
}
```

### 在后台任务中使用

```go
// internal/task/background.go
import (
    "huahua-service/pkg/leader"
    "github.com/google/uuid"
)

type Runner struct {
    lock     *leader.Lock
    leaderID string
    // ... 其他字段
}

func NewRunner(cfg Config) *Runner {
    leaderID := fmt.Sprintf("%d:%s", os.Getpid(), uuid.New().Hex())
    return &Runner{
        lock:     leader.NewLock(db.Redis, "lock:bg_refresh_loop:v1", leaderID, 180*time.Second),
        leaderID: leaderID,
    }
}

func (r *Runner) Run(ctx context.Context) {
    slog.Info("后台任务启动")
    ticker := time.NewTicker(60 * time.Second)
    defer ticker.Stop()

    // 心跳协程
    go r.heartbeat(ctx)

    for {
        select {
        case <-ticker.C:
            // 先检查是否为 leader
            acquired, err := r.lock.Acquire(ctx)
            if err != nil || !acquired {
                continue  // 不是 leader，跳过
            }
            r.tick(ctx)
        case <-ctx.Done():
            r.lock.Release(context.Background())
            slog.Info("后台任务停止")
            return
        }
    }
}

func (r *Runner) heartbeat(ctx context.Context) {
    ticker := time.NewTicker(55 * time.Second)
    defer ticker.Stop()
    for {
        select {
        case <-ticker.C:
            r.lock.Renew(ctx)
        case <-ctx.Done():
            return
        }
    }
}
```
