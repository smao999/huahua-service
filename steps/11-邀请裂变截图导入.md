```python
    # 1. 被邀请人无条件获得 5 天 VIP
    _extend_vip(new_user, INVITE_REWARD_DAYS)  # = 5 天
    
    # 2. 用 FOR UPDATE 锁行，防止并发超发
    inviter_locked = db.query(User).filter(
        User.id == inviter.id
    ).with_for_update().first()  # ← 行级锁！
    
    if inviter_locked:
        # 3. 在锁保护下统计已奖励次数
        reward_count = db.query(User).filter(
            User.invited_by == inviter_locked.id
        ).count()
        
        # 4. 如果没超上限（20 次），发奖励
        if reward_count < MAX_INVITE_REWARD_TIMES:
            _extend_vip(inviter_locked, INVITE_REWARD_DAYS)
    
    db.commit()
```
</details>


**为什么要加 `with_for_update()`？**

```
场景：两个人同时用同一个邀请码注册

时间线：
T1: A 注册 → 查 inviter 的 reward_count → count=19
T2: B 注册 → 查 inviter 的 reward_count → count=19（T1 还没提交！）
T3: A 提交 → reward_count=20 ✅
T4: B 提交 → reward_count=21 ❌ 超发了！

有 FOR UPDATE：
T1: A 注册 → FOR UPDATE 锁住 inviter 行 → count=19 → commit → 解锁
T2: B 注册 → 等 T1 解锁 → count=20 → count >= 20 → 不奖励
```

```go
// 在 repository 中提供事务方法
func (r *userRepo) LockByIDForUpdate(ctx context.Context, tx *gorm.DB, id int64) (*model.User, error) {
    var user model.User
    err := tx.WithContext(ctx).
        Raw("SELECT * FROM users WHERE id = ? FOR UPDATE", id).
        Scan(&user).Error
    return &user, err
}

// 在 service 中使用事务
func (s *AuthService) applyInviteReward(ctx context.Context, inviterID, newUserID int64) error {
    return s.db.Transaction(func(tx *gorm.DB) error {
        // 1. 锁行
        inviter, err := s.userRepo.LockByIDForUpdate(ctx, tx, inviterID)
        if err != nil {
            return err
        }
        
        // 2. 统计已奖励次数
        count, err := s.userRepo.CountInvitedBy(ctx, tx, inviter.ID)
        if err != nil {
            return err
        }
        
        // 3. 不超过上限才奖励
        if count < 20 {  // MAX_INVITE_REWARD_TIMES = 20
            inviter.VIPExpiresAt = inviter.VIPExpiresAt.Add(5 * 24 * time.Hour)
            if err := s.userRepo.Update(ctx, tx, inviter); err != nil {
                return err
            }
        }
        
        return nil // commit
    })
}
```
