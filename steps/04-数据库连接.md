# Step 4：数据库连接

> **目标**：连接 PostgreSQL 和 Redis，为后面的数据操作做准备。
> 配置项（数据库地址、密码等）放在环境变量中。

---

## 4.1 扩展 Config

在 `config.go` 添加数据库相关配置：

```go
type Config struct {
    Port  string
    DB    DatabaseConfig
    Redis RedisConfig
}

type DatabaseConfig struct {
    DSN          string // postgres://user:pass@localhost:5432/db?sslmode=disable
    MaxOpenConns int
    MaxIdleConns int
}

type RedisConfig struct {
    Addr     string // localhost:6379
    Password string
    DB       int
}

// 在 Load() 中添加：
DB: DatabaseConfig{
    DSN:          getEnv("DATABASE_URL", "postgres://localhost:5432/huahuadb?sslmode=disable"),
    MaxOpenConns: getEnvInt("DB_MAX_OPEN_CONNS", 20),
    MaxIdleConns: getEnvInt("DB_MAX_IDLE_CONNS", 10),
},
Redis: RedisConfig{
    Addr:     getEnv("REDIS_ADDR", "localhost:6379"),
    Password: getEnv("REDIS_PASSWORD", ""),
    DB:       getEnvInt("REDIS_DB", 0),
},
```

## 4.2 创建 db 包——连接 PostgreSQL 和 Redis

```go
// internal/db/db.go
package db

import (
    "context"
    "fmt"
    "log"
    "time"

    "huahua-service/internal/config"

    "github.com/redis/go-redis/v9"
    "gorm.io/driver/postgres"
    "gorm.io/gorm"
    "gorm.io/gorm/logger"
)

var (
    GORM  *gorm.DB
    Redis *redis.Client
)

func InitGORM(cfg config.DatabaseConfig) {
    db, err := gorm.Open(postgres.Open(cfg.DSN), &gorm.Config{
        Logger: logger.Default.LogMode(logger.Info),
    })
    if err != nil {
        log.Fatalf("数据库连接失败: %v", err)
    }
    sqlDB, _ := db.DB()
    sqlDB.SetMaxOpenConns(cfg.MaxOpenConns)
    sqlDB.SetMaxIdleConns(cfg.MaxIdleConns)
    sqlDB.SetConnMaxLifetime(30 * time.Minute)
    GORM = db
    log.Println("数据库已连接")
}

func InitRedis(cfg config.RedisConfig) {
    client := redis.NewClient(&redis.Options{
        Addr:     cfg.Addr,
        Password: cfg.Password,
        DB:       cfg.DB,
    })
    ctx, cancel := context.WithTimeout(context.Background(), 5*time.Second)
    defer cancel()
    if err := client.Ping(ctx).Err(); err != nil {
        log.Fatalf("Redis 连接失败: %v", err)
    }
    Redis = client
    log.Println("Redis 已连接")
}

func Close() {
    if GORM != nil {
        if sqlDB, err := GORM.DB(); err == nil {
            sqlDB.Close()
        }
    }
    if Redis != nil {
        Redis.Close()
    }
}
```

## 4.3 更新 main.go——连接数据库

```go
// cmd/server/main.go
func main() {
    cfg := config.Load()
    
    // 初始化数据库连接
    db.InitGORM(cfg.DB)
    db.InitRedis(cfg.Redis)
    defer db.Close()
    
    r := gin.Default()
    h := handler.NewHealthHandler()
    router.Register(r, h)
    r.Run(":" + cfg.Port)
}
```

## 4.4 更新 health handler——检查数据库状态

```go
// internal/handler/health.go
func (h *HealthHandler) Ping(c *gin.Context) {
    components := gin.H{}
    
    dbStatus := "ok"
    if db.GORM != nil {
        ctx, cancel := context.WithTimeout(c.Request.Context(), 2*time.Second)
        defer cancel()
        var result int
        if err := db.GORM.WithContext(ctx).Raw("SELECT 1").Scan(&result).Error; err != nil {
            dbStatus = "error"
        }
    }
    components["db"] = dbStatus
    
    redisStatus := "ok"
    if db.Redis != nil {
        ctx, cancel := context.WithTimeout(c.Request.Context(), 2*time.Second)
        defer cancel()
        if err := db.Redis.Ping(ctx).Err(); err != nil {
            redisStatus = "error"
        }
    }
    components["redis"] = redisStatus
    
    statusCode := http.StatusOK
    status := "ok"
    if dbStatus == "error" || redisStatus == "error" {
        status = "error"
        statusCode = http.StatusServiceUnavailable
    }
    
    c.JSON(statusCode, gin.H{"status": status, "components": components})
}
```
