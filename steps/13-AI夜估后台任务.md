## 13.1 AI Service

```go
// internal/service/ai/ai_service.go
package ai

import (
    "bytes"
    "context"
    "encoding/json"
    "fmt"
    "io"
    "net/http"
    "time"
)

type AIService struct {
    openRouterKey string
    geminiKey     string
    client        *http.Client
}

func NewService(openRouterKey, geminiKey string) *AIService {
    return &AIService{
        client:        &http.Client{Timeout: 90 * time.Second},
        openRouterKey: openRouterKey,
        geminiKey:     geminiKey,
    }
}

func (s *AIService) Chat(ctx context.Context, messages []Message) (string, error) {
    result, err := s.callOpenRouter(ctx, messages)
    if err == nil {
        return result, nil
    }
    return s.callGemini(ctx, messages)
}

type Message struct {
    Role    string `json:"role"`
    Content string `json:"content"`
}

func (s *AIService) callOpenRouter(ctx context.Context, messages []Message) (string, error) {
    body, _ := json.Marshal(map[string]interface{}{
        "model":    "openai/gpt-4o",
        "messages": messages,
    })
    req, _ := http.NewRequestWithContext(ctx, "POST",
        "https://openrouter.ai/api/v1/chat/completions",
        bytes.NewReader(body))
    req.Header.Set("Authorization", "Bearer "+s.openRouterKey)
    req.Header.Set("Content-Type", "application/json")
    resp, err := s.client.Do(req)
    if err != nil {
        return "", err
    }
    defer resp.Body.Close()
    var result struct {
        Choices []struct {
            Message struct {
                Content string `json:"content"`
            } `json:"message"`
        } `json:"choices"`
    }
    respBody, _ := io.ReadAll(resp.Body)
    json.Unmarshal(respBody, &result)
    if len(result.Choices) == 0 {
        return "", fmt.Errorf("no response")
    }
    return result.Choices[0].Message.Content, nil
}

func (s *AIService) callGemini(ctx context.Context, messages []Message) (string, error) {

}
```

## 13.3 后台任务

```go
// internal/task/background.go
package task

import (
    "context"
    "log"
    "time"
)

type Runner struct {
    // calibService *calibration.Service
}

func (r *Runner) Run(ctx context.Context) {
    ticker := time.NewTicker(60 * time.Second)
    defer ticker.Stop()
    for {
        select {
        case <-ticker.C:
            r.tick()
        case <-ctx.Done():
            log.Println("后台任务停止")
            return
        }
    }
}

func (r *Runner) tick() {
    now := time.Now()
    // 22:00 → 夜估校准
    if now.Hour() == 22 && now.Minute() == 0 {
        log.Println("执行夜估校准...")
    }
}
```
